All commands run in Windows cmd.exe / PowerShell, against a native-Windows
Hadoop 3.2.4 pseudo-distributed cluster (HADOOP_HOME=C:\hadoop).
Python is invoked with its full path (8.3 short form) inside -mapper /
-reducer arguments because Hadoop Streaming task containers on Windows do
not reliably inherit the full system PATH:
    C:\Progra~1\Python313\python.exe

2. MAPREDUCE JOBS (Section 8a-8i)
------------------------------------------------------------------------
:: Set JAR_PATH once per session for brevity
set JAR_PATH=C:/hadoop/share/hadoop/tools/lib/hadoop-streaming-3.2.4.jar
set PY=C:\Progra~1\Python313\python.exe

:: --- 8a. Hourly Taxi Demand ---
hdfs dfs -rm -r /taxi_project/output/hourly
hadoop jar %JAR_PATH% -files "file:///C:/taxi_project/mappers/mapper_hourly.py,file:///C:/taxi_project/reducers/reducer_hourly.py" -input /taxi_project/input/cleaned -output /taxi_project/output/hourly -mapper "%PY% mapper_hourly.py" -reducer "%PY% reducer_hourly.py"
hdfs dfs -cat /taxi_project/output/hourly/part-00000

:: --- 8b. Daily Demand ---
hdfs dfs -rm -r /taxi_project/output/daily
hadoop jar %JAR_PATH% -files "file:///C:/taxi_project/mappers/mapper_daily.py,file:///C:/taxi_project/reducers/reducer_daily.py" -input /taxi_project/input/cleaned -output /taxi_project/output/daily -mapper "%PY% mapper_daily.py" -reducer "%PY% reducer_daily.py"
hdfs dfs -cat /taxi_project/output/daily/part-00000

:: --- 8c. Pickup Location Analysis ---
hdfs dfs -rm -r /taxi_project/output/locations
hadoop jar %JAR_PATH% -files "file:///C:/taxi_project/mappers/mapper_location.py,file:///C:/taxi_project/reducers/reducer_location.py" -input /taxi_project/input/cleaned -output /taxi_project/output/locations -mapper "%PY% mapper_location.py" -reducer "%PY% reducer_location.py"
hdfs dfs -get /taxi_project/output/locations C:\taxi_project\output_local\
python C:\taxi_project\extract_top_bottom_locations.py C:\taxi_project\output_local\locations\part-00000
